2.1.4 Сортировка результатов выравнивания

В данной работе, для обеспечения высокой достоверности идентификации, был установлен строгий порог гомологии (‘pident’ ) на уровне 90%, а допустимая разница в длине последовательностей не превышала 5%.

Code 
import pandas as pd
import os
from tqdm import tqdm
import time
import re
Code 
#считываем файл с информацией про нашу базу данных
database = pd.read_csv('Obtained-data/VFDB-database/modified_VFDB_setB_full_pro.fas.fai', sep='\t', header=None,
                    names=['sseqid', 'ref_length', 'byte_start', 'bases_line', 'bytes_line'])[['sseqid', 'ref_length']]

pattern = r".*\.csv$"
regex = re.compile(pattern)
directory = 'Obtained-data/Blast-results-raw'

#обрабатываем все сырые результаты бласта, они лежат в одной папке и имеют формат .csv
for filename in tqdm(os.listdir("Obtained-data/Blast-results-raw"), desc = 'tqdm() Progress Bar'):
    time.sleep(0.5)
    if regex.match(filename):
      print(f"\n{filename}\n")
      filepath = os.path.join(directory, filename)  
      #достаем из названия файла инфу о пациенте
      patient = filename.split('file_')[1].split('.')[0]
      time.sleep(0.5)
      bl = pd.read_csv(filepath, sep="\t", header=None, 
                       names=['orf', 'sseqid' ,'pident', 'length', 'mismatch', 'gapopen', 'qstart' ,'qend', 'sstart' ,'send', 'evalue', 'bitscore']) 
      #оставляем только лучшее совпадение
      bl = bl.drop_duplicates(subset=['orf'], keep='first') 
      #ставим пороговый процент гомологии в 90%
      bl = bl[bl['pident'] >= 90]
      #объединяем с инфой о референсных последовательностях
      merged = bl.merge(database, on='sseqid', how='inner')
      #print(merged.to_markdown())
      #выбираем те выравнивания, где длины найденной и референсной последовательности отличаются не более, чем на 5%
      merged_sorted = merged.loc[(merged['ref_length'] / merged['length'] >= 0.95) & (merged['ref_length'] / merged['length'] < 1.05)]
      
      merged_sorted.to_csv(f'Blast-results/Blast-results-filtered/filtered_blast_file_{patient}.csv', index=False, sep='\t')

KRAKEN_OUTPUT_DIR="Kraken-results/Kraken_outputs"
KRAKEN_REPORT_DIR="Kraken-results/Kraken_reports"
KRAKEN_MPA_REPORT_DIR="Kraken-results/Kraken_mpa_reports"

for filename in tqdm(os.listdir(KRAKEN_OUTPUT_DIR), desc='Progress Bar'):
  
    kraken_output_path = os.path.join(KRAKEN_OUTPUT_DIR, filename)
    kraken_report_path = os.path.join(KRAKEN_REPORT_DIR, filename.replace('output','report'))
    kraken_mpa_report_path = os.path.join(KRAKEN_MPA_REPORT_DIR, filename.replace('output','mpa_report'))
    kraken_table_path = os.path.join(KRAKEN_TABLES_DIR, filename.replace('output','table'))
    
    kraken_output = pd.read_csv(kraken_output_path, sep = '\t',names = ['"C"/"U"','Contig','ID','Length','List'])[['Contig','ID']]
    kraken_report = pd.read_csv(kraken_report_path, sep = '\t', 
    names = ['Abundance','All reads','Cur reads','Level_code','ID','Name'])[['ID']].loc[2:,:].reset_index(drop = True)
    data = []
    
    with open(kraken_mpa_report_path, 'r') as f:
        for line in f:
            taxonomy_string, count = line.split('\t')
            taxonomy_ranks = taxonomy_string.split('|')
            row = {}  # Initialize the row dictionary
            for rank in taxonomy_ranks:
                rank_type, rank_name = rank.split('__', 1)  # Split at the first occurrence
                if rank_type in ['d','p','s','s1']:
                  row[rank_type] = rank_name

            data.append(row)

    taxonomy_df = pd.DataFrame(data)
    if len(taxonomy_df) != len(kraken_report):
        print(len(taxonomy_df),len(kraken_report))
        raise ValueError("The number of rows in taxonomy file does not match the number of rows in metadata file, merging cannot be performed. ")
    # 4. Merge the DataFrames based on index
    Tax_ID_df = pd.concat([kraken_report, taxonomy_df], axis=1) # Concatenate horizontally along the columns
    merged = pd.merge(Tax_ID_df,kraken_output, on = 'ID',how = 'right')
    merged = merged.dropna(axis=1, how='all')
    merged.to_csv(kraken_table_path,index = False)

2.1.6 Подсчет встречаемости ОРС (нуклеотидные последовательности)

Код запуска:

#source /Users/lidaflorenskaya/miniconda3/bin/activate py3_study
#conda install bioconda::bwa
aligned_reads_directory="Readcounts/aligned_reads"

for file in Obtained-data/ORFs/NT/*_orf_nt_names.fasta; do
    filename="${file##*/}"

    [[ "$filename" =~ ([A-Za-z]+)_([0-9]+)_orf_nt_names\.fasta ]]
    prefix="${BASH_REMATCH[1]}"
    number="${BASH_REMATCH[2]}"
    
    # Индексируем нуклеотидные последовательности ОРС
    bwa index "$file" 
    
    # Картируем риды на ОРС
    bwa mem "$file" \ 
    Obtained-data/Trimmed-reads/${prefix}_${number}_R1.fq.gz \
    Obtained-data/Trimmed-reads/${prefix}_${number}_R2.fq.gz \
    > "$aligned_reads_directory/aligned_read_${prefix}_${number}.sam"

    cd Readcounts/aligned_reads # чтобы все доп. технические файлы создавались в этой директории
    
    # Переводим SAM в BAM
    samtools view -@ 10 -b "aligned_read_${prefix}_${number}.sam" -F 0x4 | samtools sort -@ 10 - > "aligned_read_${prefix}_${number}.bam"
    
    #Индексируем BAM
    samtools index "aligned_read_${prefix}_${number}.bam"

    cd -
    
    # Подсчитываем ридкаунты и записываем результат в файл
    samtools idxstats "$aligned_reads_directory/aligned_read_${prefix}_${number}.bam" \
    > Readcounts/Readcount-result/Readcount_result_${prefix}_${number}.txt
    
    #Удалим большой SAM, который нам не нужен
    rm -r "$aligned_reads_directory/aligned_read_${prefix}_${number}.sam"
    
    
done
